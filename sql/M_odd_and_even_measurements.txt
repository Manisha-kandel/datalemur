-- M_odd_and_even_measurements, 10 minutes
WITH cte AS (
  SELECT measurement_id, measurement_value, CAST(measurement_time AS DATE) AS measurement_day,
  ROW_NUMBER() OVER(
    PARTITION BY CAST(measurement_time AS DATE) 
    ORDER BY (measurement_time) ASC
  ) AS row_num
  FROM measurements
)

SELECT 
  measurement_day, 
  SUM(CASE WHEN row_num%2 = 1 THEN measurement_value ELSE 0 END) AS odd_sum, 
  SUM(CASE WHEN row_num%2 = 0 THEN measurement_value ELSE 0 END) AS even_sum
FROM cte
GROUP BY measurement_day
;
