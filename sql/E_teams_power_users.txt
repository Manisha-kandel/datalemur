--august 2022
--top2  with highest number of messages sent
--display id, #messages
--output the results in descending order
--distinct #messages for all: row number, rank and dense_rank can all be used.

SELECT sender_id, COUNT(*) as message_count 
FROM messages
WHERE 
EXTRACT(MONTH FROM sent_date) = 8 AND EXTRACT(YEAR FROM sent_date) = 2022
GROUP BY sender_id
ORDER BY message_count DESC 
LIMIT 2
;
