-- M_international_call_percentage, 10 minutes
-- use conditional aggreation 

WITH cte AS (
SELECT 
  c.*, 
  i1.country_id AS caller_country, 
  i2.country_id AS receiver_country
FROM phone_calls c 
LEFT JOIN phone_info i1  
  ON c.caller_id = i1.caller_id 
LEFT JOIN phone_info i2
  ON c.receiver_id = i2.caller_id
),  

cte2 AS (
SELECT 
  COUNT(*) AS total_calls,
  SUM(CASE WHEN caller_country <> receiver_country THEN 1 ELSE 0 END) AS international_calls 
FROM cte
)

SELECT ROUND((international_calls * 100.0 / total_calls), 1) AS international_calls_pct
FROM cte2
;
